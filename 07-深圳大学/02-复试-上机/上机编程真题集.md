### 题目1.1：统计每个数出现的次数并按次数排序

**题目描述**
给定n个整数，请统计出每个整数出现的次数，按出现次数从多到少的顺序输出各个整数及其次数。如果次数相同，按数值从小到大输出。

**输入格式**
第一行：一个整数n (1 ≤ n ≤ 100000)
第二行：n个整数

**输出格式**
每行输出两个整数：数值和次数

**输入样例**
```
10
3 1 2 3 2 3 4 1 5 1
```

**输出样例**
```
1 3
3 3
2 2
4 1
5 1
```

**解题思路**
1. 使用哈希表统计每个数出现的次数
2. 将统计结果存入vector，自定义排序：次数降序，数值升序

**参考代码 (C++)**
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <unordered_map>
using namespace std;

bool compare(const pair<int, int>& a, const pair<int, int>& b) {
    if (a.second != b.second) return a.second > b.second;
    return a.first < b.first;
}

int main() {
    int n;
    scanf("%d", &n);
    unordered_map<int, int> count;
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        count[x]++;
    }
    vector<pair<int, int>> result(count.begin(), count.end());
    sort(result.begin(), result.end(), compare);
    for (auto& p : result) {
        printf("%d %d\n", p.first, p.second);
    }
    return 0;
}
```

### 题目1.2：归并排序求逆序数

**题目描述**
给定一个n个元素的序列，求其逆序数（逆序对：i<j但a[i]>a[j]的对数）。

**输入格式**
第一行：整数n (1 ≤ n ≤ 100000)
第二行：n个整数

**输出格式**
输出一个整数，表示逆序数

**输入样例**
```
5
3 1 4 5 2
```

**输出样例**
```
5
```

**解题思路**
在归并排序的合并过程中，当右半部分元素小于左半部分当前元素时，左半部分剩余元素都与该右元素构成逆序对。

**参考代码 (C)**
```c
#include <stdio.h>
#include <stdlib.h>

long long count = 0;
int temp[100005];

void merge(int arr[], int left, int mid, int right) {
    int i = left, j = mid + 1, k = left;
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            temp[k++] = arr[j++];
            count += (mid - i + 1);
        }
    }
    while (i <= mid) temp[k++] = arr[i++];
    while (j <= right) temp[k++] = arr[j++];
    for (i = left; i <= right; i++) arr[i] = temp[i];
}

void mergeSort(int arr[], int left, int right) {
    if (left < right) {
        int mid = (left + right) / 2;
        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

int main() {
    int n;
    scanf("%d", &n);
    int* arr = (int*)malloc(sizeof(int) * n);
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    mergeSort(arr, 0, n - 1);
    printf("%lld\n", count);
    free(arr);
    return 0;
}
```

### 题目1.3：第k大元素

**题目描述**
给定一个整数数组，找出数组中第k大的元素。

**输入格式**
第一行：整数n和k
第二行：n个整数

**输出格式**
第k大的元素

**输入样例**
```
6 3
3 2 1 5 6 4
```

**输出样例**
```
4
```

**解题思路**
快速选择算法（快排变种），平均O(n)，最坏O(n²)

**参考代码 (C++)**
```cpp
#include <iostream>
#include <algorithm>
using namespace std;

int quickSelect(int nums[], int left, int right, int k) {
    if (left == right) return nums[left];
    int pivot = nums[right];
    int i = left - 1;
    for (int j = left; j < right; j++) {
        if (nums[j] <= pivot) {
            i++;
            swap(nums[i], nums[j]);
        }
    }
    swap(nums[i + 1], nums[right]);
    int pivotIndex = i + 1;
    if (k == pivotIndex) return nums[k];
    else if (k < pivotIndex) return quickSelect(nums, left, pivotIndex - 1, k);
    else return quickSelect(nums, pivotIndex + 1, right, k);
}

int main() {
    int n, k;
    cin >> n >> k;
    int* nums = new int[n];
    for (int i = 0; i < n; i++) cin >> nums[i];
    cout << quickSelect(nums, 0, n - 1, n - k) << endl;
    delete[] nums;
    return 0;
}
```

### 题目1.4：合并k个有序数组

**题目描述**
给定k个有序数组，将它们合并成一个有序数组并输出。

**输入格式**
第一行：整数k
接下来k行：每行第一个整数是n，然后是n个有序整数

**输出格式**
合并后的有序数组

**输入样例**
```
3
3 1 4 7
2 2 5
4 3 6 8 9
```

**输出样例**
```
1 2 3 4 5 6 7 8 9
```

**解题思路**
使用最小堆，每次从k个数组中选最小的元素，O(Nlogk)

**参考代码 (C++)**
```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct Element {
    int value;
    int arrayIndex;
    int elementIndex;
    Element(int v, int ai, int ei) : value(v), arrayIndex(ai), elementIndex(ei) {}
    bool operator>(const Element& other) const { return value > other.value; }
};

int main() {
    int k;
    cin >> k;
    vector<vector<int>> arrays(k);
    for (int i = 0; i < k; i++) {
        int n;
        cin >> n;
        arrays[i].resize(n);
        for (int j = 0; j < n; j++) cin >> arrays[i][j];
    }
    priority_queue<Element, vector<Element>, greater<Element>> minHeap;
    for (int i = 0; i < k; i++) {
        if (!arrays[i].empty()) minHeap.push(Element(arrays[i][0], i, 0));
    }
    vector<int> result;
    while (!minHeap.empty()) {
        Element curr = minHeap.top();
        minHeap.pop();
        result.push_back(curr.value);
        if (curr.elementIndex + 1 < arrays[curr.arrayIndex].size()) {
            int next = curr.elementIndex + 1;
            minHeap.push(Element(arrays[curr.arrayIndex][next], curr.arrayIndex, next));
        }
    }
    for (int i = 0; i < result.size(); i++) {
        if (i > 0) cout << " ";
        cout << result[i];
    }
    cout << endl;
    return 0;
}
```

### 题目1.5：三色排序（荷兰国旗问题）

**题目描述**
给定包含0、1、2的数组，原地排序使相同颜色相邻，按0、1、2顺序排列。

**输入格式**
第一行：整数n
第二行：n个整数

**输出格式**
排序后的数组

**输入样例**
```
8
2 0 2 1 1 0 2 1
```

**输出样例**
```
0 0 1 1 1 2 2 2
```

**解题思路**
三个指针：low（0区末尾）、mid（当前扫描）、high（2区开头）

**参考代码 (C++)**
```cpp
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    int* nums = new int[n];
    for (int i = 0; i < n; i++) cin >> nums[i];
    int low = 0, mid = 0, high = n - 1;
    while (mid <= high) {
        if (nums[mid] == 0) {
            swap(nums[low], nums[mid]);
            low++; mid++;
        } else if (nums[mid] == 1) {
            mid++;
        } else {
            swap(nums[mid], nums[high]);
            high--;
        }
    }
    for (int i = 0; i < n; i++) {
        if (i > 0) cout << " ";
        cout << nums[i];
    }
    cout << endl;
    delete[] nums;
    return 0;
}
```

**解题思路**
迭代法：三个指针prev, curr, next逐个反转

**参考代码 (C)**
```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode* prev = NULL;
    struct ListNode* curr = head;
    while (curr != NULL) {
        struct ListNode* next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    return prev;
}

int main() {
    int n;
    scanf("%d", &n);
    struct ListNode dummy;
    dummy.next = NULL;
    struct ListNode* tail = &dummy;
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        struct ListNode* node = (struct ListNode*)malloc(sizeof(struct ListNode));
        node->val = x; node->next = NULL;
        tail->next = node; tail = node;
    }
    struct ListNode* newHead = reverseList(dummy.next);
    struct ListNode* curr = newHead;
    int first = 1;
    while (curr != NULL) {
        if (!first) printf(" ");
        printf("%d", curr->val);
        first = 0; curr = curr->next;
    }
    printf("\n");
    return 0;
}
```

### 题目2.2：判断链表是否有环并找入口

**题目描述**
给定一个单链表，判断是否存在环。如果有环，找出入口节点的值；否则输出-1。

**输入格式**
第一行：n
第二行：n个整数
第三行：整数p（最后一个节点next指向第p个节点，p=0表示无环）

**输出格式**
入口节点的值或-1

**输入样例**
```
5
3 2 0 -4 7
2
```

**输出样例**
```
2
```

**解题思路**
快慢指针找相遇点，然后一个从head一个从相遇点同步走，相遇点即入口

**参考代码 (C++)**
```cpp
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(NULL) {}
};

ListNode* detectCycle(ListNode* head) {
    if (head == NULL || head->next == NULL) return NULL;
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) break;
    }
    if (fast == NULL || fast->next == NULL) return NULL;
    ListNode* ptr = head;
    while (ptr != slow) {
        ptr = ptr->next;
        slow = slow->next;
    }
    return ptr;
}

int main() {
    int n;
    cin >> n;
    ListNode** nodes = new ListNode*[n + 1];
    for (int i = 1; i <= n; i++) {
        int val; cin >> val;
        nodes[i] = new ListNode(val);
    }
    for (int i = 1; i < n; i++) nodes[i]->next = nodes[i + 1];
    int p; cin >> p;
    if (p > 0) nodes[n]->next = nodes[p];
    ListNode* cycleEntry = detectCycle(nodes[1]);
    if (cycleEntry != NULL) cout << cycleEntry->val << endl;
    else cout << -1 << endl;
    return 0;
}
```

### 题目2.3：合并两个有序链表

**题目描述**
将两个升序链表合并为一个新的升序链表并返回。

**输入格式**
第一行：n和m
第二行：n个整数
第三行：m个整数

**输出格式**
合并后的有序链表

**输入样例**
```
3 3
1 2 4
1 3 4
```

**输出样例**
```
1 1 2 3 4 4
```

**解题思路**
使用虚拟头节点，依次比较两个链表当前节点

**参考代码 (C)**
```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

struct ListNode* mergeTwoLists(struct ListNode* l1, struct ListNode* l2) {
    struct ListNode dummy;
    dummy.next = NULL;
    struct ListNode* curr = &dummy;
    while (l1 != NULL && l2 != NULL) {
        if (l1->val <= l2->val) {
            curr->next = l1; l1 = l1->next;
        } else {
            curr->next = l2; l2 = l2->next;
        }
        curr = curr->next;
    }
    if (l1 != NULL) curr->next = l1;
    if (l2 != NULL) curr->next = l2;
    return dummy.next;
}

struct ListNode* createList(int* arr, int n) {
    struct ListNode dummy;
    dummy.next = NULL;
    struct ListNode* tail = &dummy;
    for (int i = 0; i < n; i++) {
        struct ListNode* node = (struct ListNode*)malloc(sizeof(struct ListNode));
        node->val = arr[i]; node->next = NULL;
        tail->next = node; tail = node;
    }
    return dummy.next;
}

int main() {
    int n, m;
    scanf("%d %d", &n, &m);
    int* arr1 = (int*)malloc(sizeof(int) * n);
    int* arr2 = (int*)malloc(sizeof(int) * m);
    for (int i = 0; i < n; i++) scanf("%d", &arr1[i]);
    for (int i = 0; i < m; i++) scanf("%d", &arr2[i]);
    struct ListNode* l1 = createList(arr1, n);
    struct ListNode* l2 = createList(arr2, m);
    struct ListNode* merged = mergeTwoLists(l1, l2);
    struct ListNode* curr = merged;
    int first = 1;
    while (curr != NULL) {
        if (!first) printf(" ");
        printf("%d", curr->val);
        first = 0; curr = curr->next;
    }
    printf("\n");
    return 0;
}
```

### 题目2.4：判断回文链表

**题目描述**
给定一个单链表，先找到中间节点，然后判断是否为回文链表。

**输入格式**
第一行：n
第二行：n个整数

**输出格式**
第一行：中间节点值
第二行：YES或NO

**输入样例**
```
5
1 2 3 2 1
```

**输出样例**
```
3
YES
```

**解题思路**
1. 快慢指针找中间节点
2. 反转后半部分链表，与前半部分比较

**参考代码 (C++)**
```cpp
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(NULL) {}
};

ListNode* findMiddle(ListNode* head) {
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
    }
    return slow;
}

ListNode* reverseList(ListNode* head) {
    ListNode* prev = NULL;
    ListNode* curr = head;
    while (curr != NULL) {
        ListNode* next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    return prev;
}

bool isPalindrome(ListNode* head) {
    if (head == NULL || head->next == NULL) return true;
    ListNode* mid = findMiddle(head);
    ListNode* secondHalf = reverseList(mid);
    ListNode* firstHalf = head;
    while (secondHalf != NULL) {
        if (firstHalf->val != secondHalf->val) return false;
        firstHalf = firstHalf->next;
        secondHalf = secondHalf->next;
    }
    return true;
}

int main() {
    int n;
    cin >> n;
    ListNode dummy(0);
    ListNode* tail = &dummy;
    for (int i = 0; i < n; i++) {
        int val; cin >> val;
        tail->next = new ListNode(val);
        tail = tail->next;
    }
    ListNode* head = dummy.next;
    ListNode* mid = findMiddle(head);
    cout << mid->val << endl;
    if (isPalindrome(head)) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
```

**输出格式**
前三行：前序、中序、后序遍历

```
5

4 0 0

```

**输出样例**


**解题思路**
递归遍历，注意顺序区别

**参考代码 (C)**
```c
#include <stdio.h>

struct TreeNode {
    int val; int left; int right;
} tree[1005];

void preorder(int root) {
    if (root == 0) return;
    printf("%d ", tree[root].val);
    preorder(tree[root].left);
    preorder(tree[root].right);
}

void inorder(int root) {
    if (root == 0) return;
    inorder(tree[root].left);
    printf("%d ", tree[root].val);
    inorder(tree[root].right);
}

void postorder(int root) {
    if (root == 0) return;
    postorder(tree[root].left);
    postorder(tree[root].right);
    printf("%d ", tree[root].val);
}

int main() {
    int n;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%d %d %d", &tree[i].val, &tree[i].left, &tree[i].right);
    }
    preorder(1); printf("\n");
    inorder(1); printf("\n");
    postorder(1); printf("\n");
    return 0;
}
```

### 题目3.2：根据前序和中序构造二叉树

**题目描述**
给定前序和中序遍历结果，构造二叉树并输出后序遍历。

**输入格式**
第一行：n
第二行：前序遍历
第三行：中序遍历

**输出格式**
后序遍历

**输入样例**
```
5
1 2 4 5 3
4 2 5 1 3
```

**输出样例**
```
4 5 2 3 1
```

**解题思路**
前序第一个是根，在中序中找到根，分成左右子树递归

**参考代码 (C++)**
```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(NULL), right(NULL) {}
};

unordered_map<int, int> inorderIndex;

TreeNode* build(vector<int>& preorder, vector<int>& inorder,
                int preStart, int preEnd, int inStart, int inEnd) {
    if (preStart > preEnd) return NULL;
    int rootVal = preorder[preStart];
    TreeNode* root = new TreeNode(rootVal);
    int rootIndex = inorderIndex[rootVal];
    int leftSize = rootIndex - inStart;
    root->left = build(preorder, inorder, preStart + 1, preStart + leftSize, inStart, rootIndex - 1);
    root->right = build(preorder, inorder, preStart + leftSize + 1, preEnd, rootIndex + 1, inEnd);
    return root;
}

void postorder(TreeNode* root) {
    if (root == NULL) return;
    postorder(root->left);
    postorder(root->right);
    cout << root->val << " ";
}

int main() {
    int n;
    cin >> n;
    vector<int> preorder(n), inorder(n);
    for (int i = 0; i < n; i++) cin >> preorder[i];
    for (int i = 0; i < n; i++) {
        cin >> inorder[i];
        inorderIndex[inorder[i]] = i;
    }
    TreeNode* root = build(preorder, inorder, 0, n - 1, 0, n - 1);
    postorder(root);
    cout << endl;
    return 0;
}
```

### 题目3.4：判断BST和平衡二叉树

**题目描述**
判断是否为二叉搜索树和平衡二叉树。

**输入格式**
同题目3.1

**输出格式**
第一行：YES或NO（BST）
第二行：YES或NO（平衡）

**输入样例**
```
5
4 2 5
2 1 3
5 0 0
1 0 0
3 0 0
```

**输出样例**
```
YES
YES
```

**解题思路**
1. BST：中序遍历应该严格递增
2. 平衡：每个节点左右子树高度差不超过1

**参考代码 (C++)**
```cpp
#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
using namespace std;

struct Node { int val; int left; int right; } tree[1005];

void inorder(int root, vector<int>& result) {
    if (root == 0) return;
    inorder(tree[root].left, result);
    result.push_back(tree[root].val);
    inorder(tree[root].right, result);
}

bool isBST(int root) {
    vector<int> seq;
    inorder(root, seq);
    for (int i = 1; i < seq.size(); i++) {
        if (seq[i] <= seq[i - 1]) return false;
    }
    return true;
}

int checkBalance(int root) {
    if (root == 0) return 0;
    int left = checkBalance(tree[root].left);
    if (left == -1) return -1;
    int right = checkBalance(tree[root].right);
    if (right == -1) return -1;
    if (abs(left - right) > 1) return -1;
    return max(left, right) + 1;
}

bool isBalanced(int root) {
    return checkBalance(root) != -1;
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> tree[i].val >> tree[i].left >> tree[i].right;
    }
    if (isBST(1)) cout << "YES" << endl;
    else cout << "NO" << endl;
    if (isBalanced(1)) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
```

### 题目4.3：Dijkstra单源最短路径

**题目描述**
给定带权有向图，求从节点1到其他所有节点的最短路径。

**输入格式**
第一行：n和m
接下来m行：u v w

**输出格式**
n-1行：节点: 最短距离

**输入样例**
```
5 8
1 2 10
1 4 30
1 5 100
2 3 50
3 5 10
4 3 20
4 5 60
3 4 20
```

**输出样例**
```
2: 10
3: 50
4: 30
5: 60
```

**解题思路**
优先队列优化的Dijkstra算法 O(mlogn)

**参考代码 (C++)**
```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;

const int INF = INT_MAX;
vector<pair<int, int>> adj[1005];
int dist[1005];
int n, m;

void dijkstra(int start) {
    for (int i = 1; i <= n; i++) dist[i] = INF;
    dist[start] = 0;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, start});
    while (!pq.empty()) {
        int u = pq.top().second;
        int d = pq.top().first;
        pq.pop();
        if (d > dist[u]) continue;
        for (auto& edge : adj[u]) {
            int v = edge.first, w = edge.second;
            if (dist[u] != INF && dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
    }
    dijkstra(1);
    for (int i = 2; i <= n; i++) {
        if (dist[i] == INF) cout << i << ": -1" << endl;
        else cout << i << ": " << dist[i] << endl;
    }
    return 0;
}
```

### 题目4.4：Floyd多源最短路径

**题目描述**
给定带权有向图，求任意两点间的最短路径。

**输入格式**
第一行：n
接下来n行：每行n个整数（0表示自己，-1表示不可达）

**输出格式**
n行n列的最短距离矩阵

**输入样例**
```
4
0 2 -1 5
-1 0 3 -1
-1 -1 0 1
-1 -1 -1 0
```

**输出样例**
```
0 2 5 5
-1 0 3 4
-1 -1 0 1
-1 -1 -1 0
```

**解题思路**
动态规划：dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])

**参考代码 (C++)**
```cpp
#include <iostream>
using namespace std;

const int INF = 1e9;
int dist[105][105];
int n;

void floyd() {
    for (int k = 1; k <= n; k++) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                if (dist[i][k] != INF && dist[k][j] != INF) {
                    if (dist[i][k] + dist[k][j] < dist[i][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> dist[i][j];
            if (dist[i][j] == -1 && i != j) dist[i][j] = INF;
        }
    }
    floyd();
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (j > 1) cout << " ";
            if (dist[i][j] >= INF) cout << -1;
            else cout << dist[i][j];
        }
        cout << endl;
    }
    return 0;
}
```

```cpp
#include <iostream>
#include <string>
using namespace std;

string expand(string s, int left, int right) {
    while (left >= 0 && right < s.length() && s[left] == s[right]) {
        left--; right++;
    }
    return s.substr(left + 1, right - left - 1);
}

string longestPalindrome(string s) {
    if (s.empty()) return "";
    string result = "";
    for (int i = 0; i < s.length(); i++) {
        string odd = expand(s, i, i);
        string even = expand(s, i, i + 1);
        if (odd.length() > result.length()) result = odd;
        if (even.length() > result.length()) result = even;
    }
    return result;
}
```

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

vector<int> computeNext(string pattern) {
    int n = pattern.length();
    vector<int> next(n, 0);
    int j = 0;
    for (int i = 1; i < n; i++) {
        while (j > 0 && pattern[i] != pattern[j]) j = next[j - 1];
        if (pattern[i] == pattern[j]) j++;
        next[i] = j;
    }
    return next;
}

vector<int> kmpSearch(string text, string pattern) {
    vector<int> positions;
    vector<int> next = computeNext(pattern);
    int j = 0;
    for (int i = 0; i < text.length(); i++) {
        while (j > 0 && text[i] != pattern[j]) j = next[j - 1];
        if (text[i] == pattern[j]) j++;
        if (j == pattern.length()) {
            positions.push_back(i - pattern.length() + 2);
            j = next[j - 1];
        }
    }
    return positions;
}
```

```cpp
#include <iostream>
#include <string>
#include <climits>
using namespace std;

int myAtoi(string s) {
    int i = 0, sign = 1, result = 0;
    int n = s.length();
    while (i < n && s[i] == ' ') i++;  // 跳过空格
    if (i < n && (s[i] == '+' || s[i] == '-')) {
        sign = (s[i] == '-') ? -1 : 1;
        i++;
    }
    while (i < n && isdigit(s[i])) {
        int digit = s[i] - '0';
        if (result > INT_MAX / 10 || (result == INT_MAX / 10 && digit > 7)) {
            return (sign == 1) ? INT_MAX : INT_MIN;
        }
        result = result * 10 + digit;
        i++;
    }
    return result * sign;
}
```

**参考代码 (C++)**
```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

string addStrings(string num1, string num2) {
    string result = "";
    int i = num1.length() - 1;
    int j = num2.length() - 1;
    int carry = 0;
    while (i >= 0 || j >= 0 || carry > 0) {
        int sum = carry;
        if (i >= 0) sum += num1[i--] - '0';
        if (j >= 0) sum += num2[j--] - '0';
        carry = sum / 10;
        result.push_back(sum % 10 + '0');
    }
    reverse(result.begin(), result.end());
    return result;
}
```
